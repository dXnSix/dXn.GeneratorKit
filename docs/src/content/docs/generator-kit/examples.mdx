# Examples

This page provides comprehensive examples showing how to use Six.GeneratorKit in real-world scenarios.

## Example 1: Property Change Notification Generator

Generate `INotifyPropertyChanged` implementations for classes.

### Attribute Definition

```csharp
using System;

namespace PropertyChangeGenerator
{
    [AttributeUsage(AttributeTargets.Class)]
    public class NotifyPropertyChangedAttribute : Attribute
    {
        public bool GenerateSetters { get; set; } = true;
        public string? BaseClassName { get; set; }
    }
}
```

### Source Generator

```csharp
using System.Collections.Immutable;
using System.Linq;
using System.Text;
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp.Syntax;
using Six.GeneratorKit;

[Generator]
public class PropertyChangeGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        var classDeclarations = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "PropertyChangeGenerator.NotifyPropertyChangedAttribute",
                predicate: static (node, _) => node is ClassDeclarationSyntax,
                transform: static (ctx, _) => GetClassToGenerate(ctx))
            .Where(static m => m is not null);

        context.RegisterSourceOutput(classDeclarations, 
            static (spc, source) => Execute(source!, spc));
    }

    private static ClassToGenerate? GetClassToGenerate(GeneratorAttributeSyntaxContext context)
    {
        if (context.TargetNode is not ClassDeclarationSyntax classDeclaration)
            return null;

        if (context.TargetSymbol is not INamedTypeSymbol classSymbol)
            return null;

        // Check if class is partial
        if (!classSymbol.IsPartial())
        {
            GeneratorService.Add(Diagnostics.ClassMustBePartial, 
                context.TargetNode.GetLocation());
            return null;
        }

        var attribute = context.Attributes[0];
        var generateSetters = attribute.GetNamedArgument("GenerateSetters", true);
        var baseClassName = attribute.GetNamedArgument("BaseClassName", "");

        // Get all auto-properties that need notification
        using var propertiesBuilder = ImmutableArrayBuilder<PropertyInfo>.Rent();
        
        foreach (var member in classSymbol.GetMembers().OfType<IPropertySymbol>())
        {
            if (ShouldGenerateForProperty(member))
            {
                propertiesBuilder.Add(new PropertyInfo(
                    member.Name,
                    member.Type.GetFullyQualifiedNameWithNullabilityAnnotations(),
                    member.GetFormattedAccessModifier(),
                    HasBackingField(member)
                ));
            }
        }

        return new ClassToGenerate(
            classSymbol.Name,
            classSymbol.GetNamespace(),
            classSymbol.GetFormattedAccessModifier(),
            generateSetters,
            baseClassName,
            propertiesBuilder.ToImmutable()
        );
    }

    private static bool ShouldGenerateForProperty(IPropertySymbol property)
    {
        // Only generate for auto-properties with both getter and setter
        return property.DeclaredAccessibility == Accessibility.Public &&
               property.GetMethod is not null &&
               property.SetMethod is not null &&
               !property.IsStatic;
    }

    private static bool HasBackingField(IPropertySymbol property)
    {
        // Check if property has auto-generated backing field
        return property.SetMethod?.IsInitOnly == false;
    }

    private static void Execute(ClassToGenerate classToGenerate, SourceProductionContext context)
    {
        try
        {
            var source = GenerateNotifyPropertyChangedClass(classToGenerate);
            var fileName = $"{classToGenerate.Name}.PropertyChanged.g.cs";
            
            var header = CodeHeaderHelpers.GetGeneratedCodeHeader("PropertyChangeGenerator");
            var fullSource = $"{header}\n{source}";
            
            context.AddSource(fileName, fullSource);
        }
        catch (System.Exception)
        {
            GeneratorService.Add(Diagnostics.GenerationError, Location.None);
        }
    }

    private static string GenerateNotifyPropertyChangedClass(ClassToGenerate classInfo)
    {
        var sb = new StringBuilder();

        // Usings
        sb.AppendLine("using System.ComponentModel;");
        sb.AppendLine("using System.Runtime.CompilerServices;");
        sb.AppendLine();

        // Namespace
        if (!string.IsNullOrEmpty(classInfo.Namespace))
        {
            sb.AppendLine($"namespace {classInfo.Namespace};");
            sb.AppendLine();
        }

        // Class declaration
        sb.AppendLine($"{classInfo.AccessModifier} partial class {classInfo.Name} : INotifyPropertyChanged");
        sb.AppendLine("{");

        // PropertyChanged event
        sb.AppendLine("    public event PropertyChangedEventHandler? PropertyChanged;");
        sb.AppendLine();

        // OnPropertyChanged method
        sb.AppendLine("    protected virtual void OnPropertyChanged([CallerMemberName] string? propertyName = null)");
        sb.AppendLine("    {");
        sb.AppendLine("        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));");
        sb.AppendLine("    }");
        sb.AppendLine();

        // SetProperty helper method
        sb.AppendLine("    protected bool SetProperty<T>(ref T field, T value, [CallerMemberName] string? propertyName = null)");
        sb.AppendLine("    {");
        sb.AppendLine("        if (EqualityComparer<T>.Default.Equals(field, value))");
        sb.AppendLine("            return false;");
        sb.AppendLine();
        sb.AppendLine("        field = value;");
        sb.AppendLine("        OnPropertyChanged(propertyName);");
        sb.AppendLine("        return true;");
        sb.AppendLine("    }");

        // Generate backing fields and properties if needed
        if (classInfo.GenerateSetters)
        {
            sb.AppendLine();
            GeneratePropertiesWithBackingFields(sb, classInfo.Properties);
        }

        sb.AppendLine("}");
        
        return sb.ToString();
    }

    private static void GeneratePropertiesWithBackingFields(StringBuilder sb, EquatableArray<PropertyInfo> properties)
    {
        foreach (var property in properties)
        {
            var backingFieldName = $"_{char.ToLowerInvariant(property.Name[0])}{property.Name[1..]}";
            
            // Backing field
            sb.AppendLine($"    private {property.Type} {backingFieldName};");
            
            // Property with notification
            sb.AppendLine($"    {property.AccessModifier} {property.Type} {property.Name}");
            sb.AppendLine("    {");
            sb.AppendLine($"        get => {backingFieldName};");
            sb.AppendLine($"        set => SetProperty(ref {backingFieldName}, value);");
            sb.AppendLine("    }");
            sb.AppendLine();
        }
    }
}

// Data models
public record ClassToGenerate(
    string Name,
    string Namespace,
    string AccessModifier,
    bool GenerateSetters,
    string BaseClassName,
    EquatableArray<PropertyInfo> Properties);

public record PropertyInfo(
    string Name,
    string Type,
    string AccessModifier,
    bool HasBackingField);

// Diagnostics
public static class Diagnostics
{
    public static readonly DiagnosticDescriptor ClassMustBePartial = new(
        id: "PCG001",
        title: "Class must be partial",
        messageFormat: "Class '{0}' must be declared as partial to generate property change notifications",
        category: "PropertyChangeGenerator",
        DiagnosticSeverity.Error,
        isEnabledByDefault: true);

    public static readonly DiagnosticDescriptor GenerationError = new(
        id: "PCG002",
        title: "Generation error",
        messageFormat: "An error occurred during code generation",
        category: "PropertyChangeGenerator",
        DiagnosticSeverity.Error,
        isEnabledByDefault: true);
}
```

## Example 2: DTO Mapper Generator

Generate mapping methods between DTOs and domain objects.

### Attribute Definition

```csharp
[AttributeUsage(AttributeTargets.Class)]
public class GenerateMapperAttribute : Attribute
{
    public Type? TargetType { get; set; }
    public bool GenerateReverseMapping { get; set; } = true;
    public string[]? IgnoreProperties { get; set; }
}
```

### Source Generator

```csharp
[Generator]
public class MapperGenerator : IIncrementalGenerator
{
    public void Initialize(IncrementalGeneratorInitializationContext context)
    {
        var mapperClasses = context.SyntaxProvider
            .ForAttributeWithMetadataName(
                "MapperGenerator.GenerateMapperAttribute",
                predicate: static (node, _) => node is ClassDeclarationSyntax,
                transform: static (ctx, _) => GetMapperInfo(ctx))
            .Where(static m => m is not null);

        context.RegisterSourceOutput(mapperClasses, 
            static (spc, source) => Execute(source!, spc));
    }

    private static MapperInfo? GetMapperInfo(GeneratorAttributeSyntaxContext context)
    {
        if (context.TargetSymbol is not INamedTypeSymbol sourceType)
            return null;

        var attribute = context.Attributes[0];
        
        // Get target type from attribute
        var targetTypeArg = attribute.GetConstructorArgument<INamedTypeSymbol>(0, null);
        if (targetTypeArg is null)
        {
            GeneratorService.Add(Diagnostics.TargetTypeRequired, 
                context.TargetNode.GetLocation());
            return null;
        }

        var generateReverse = attribute.GetNamedArgument("GenerateReverseMapping", true);
        var ignoreProps = attribute.GetConstructorArguments<string>().ToArray();

        // Analyze properties for mapping
        var sourceMappings = AnalyzeMappings(sourceType, targetTypeArg, ignoreProps);
        var targetMappings = generateReverse ? 
            AnalyzeMappings(targetTypeArg, sourceType, ignoreProps) : 
            EquatableArray<PropertyMapping>.Empty;

        return new MapperInfo(
            sourceType.Name,
            sourceType.GetNamespace(),
            sourceType.GetFormattedAccessModifier(),
            targetTypeArg.Name,
            targetTypeArg.GetNamespace(),
            generateReverse,
            sourceMappings,
            targetMappings
        );
    }

    private static EquatableArray<PropertyMapping> AnalyzeMappings(
        INamedTypeSymbol sourceType, 
        INamedTypeSymbol targetType,
        string[] ignoreProperties)
    {
        using var builder = ImmutableArrayBuilder<PropertyMapping>.Rent();

        var sourceProperties = sourceType.GetMembers()
            .OfType<IPropertySymbol>()
            .Where(p => p.GetMethod is not null)
            .ToDictionary(p => p.Name, p => p);

        var targetProperties = targetType.GetMembers()
            .OfType<IPropertySymbol>()
            .Where(p => p.SetMethod is not null)
            .Where(p => !ignoreProperties.Contains(p.Name));

        foreach (var targetProp in targetProperties)
        {
            if (sourceProperties.TryGetValue(targetProp.Name, out var sourceProp))
            {
                var mappingType = GetMappingType(sourceProp.Type, targetProp.Type);
                
                builder.Add(new PropertyMapping(
                    sourceProp.Name,
                    targetProp.Name,
                    sourceProp.Type.GetFullyQualifiedNameWithNullabilityAnnotations(),
                    targetProp.Type.GetFullyQualifiedNameWithNullabilityAnnotations(),
                    mappingType
                ));
            }
        }

        return builder.ToImmutable();
    }

    private static MappingType GetMappingType(ITypeSymbol sourceType, ITypeSymbol targetType)
    {
        if (SymbolEqualityComparer.Default.Equals(sourceType, targetType))
            return MappingType.Direct;

        if (sourceType.IsCollection() && targetType.IsCollection())
            return MappingType.Collection;

        // Check for implicit conversions
        // This is simplified - in reality you'd check for conversion operators
        if (CanConvert(sourceType, targetType))
            return MappingType.Convertible;

        return MappingType.Custom;
    }

    private static bool CanConvert(ITypeSymbol from, ITypeSymbol to)
    {
        // Simplified conversion checking
        // In a real implementation, you'd check for implicit operators
        return from.SpecialType != SpecialType.None && 
               to.SpecialType != SpecialType.None;
    }

    private static void Execute(MapperInfo mapperInfo, SourceProductionContext context)
    {
        var source = GenerateMapperClass(mapperInfo);
        var fileName = $"{mapperInfo.SourceTypeName}.Mapper.g.cs";
        
        var header = CodeHeaderHelpers.GetGeneratedCodeHeader("MapperGenerator");
        var fullSource = $"{header}\n{source}";
        
        context.AddSource(fileName, fullSource);
    }

    private static string GenerateMapperClass(MapperInfo info)
    {
        var sb = new StringBuilder();

        if (!string.IsNullOrEmpty(info.SourceNamespace))
        {
            sb.AppendLine($"namespace {info.SourceNamespace};");
            sb.AppendLine();
        }

        sb.AppendLine($"{info.AccessModifier} static partial class {info.SourceTypeName}Mapper");
        sb.AppendLine("{");

        // Generate ToTarget method
        GenerateToTargetMethod(sb, info);

        // Generate FromTarget method if reverse mapping is enabled
        if (info.GenerateReverseMapping)
        {
            sb.AppendLine();
            GenerateFromTargetMethod(sb, info);
        }

        sb.AppendLine("}");
        return sb.ToString();
    }

    private static void GenerateToTargetMethod(StringBuilder sb, MapperInfo info)
    {
        sb.AppendLine($"    public static {info.TargetTypeName} To{info.TargetTypeName}(this {info.SourceTypeName} source)");
        sb.AppendLine("    {");
        sb.AppendLine("        if (source == null) throw new ArgumentNullException(nameof(source));");
        sb.AppendLine();
        sb.AppendLine($"        return new {info.TargetTypeName}");
        sb.AppendLine("        {");

        for (int i = 0; i < info.SourceMappings.Count; i++)
        {
            var mapping = info.SourceMappings[i];
            var assignment = GeneratePropertyAssignment(mapping);
            sb.Append($"            {mapping.TargetProperty} = {assignment}");
            
            if (i < info.SourceMappings.Count - 1)
                sb.AppendLine(",");
            else
                sb.AppendLine();
        }

        sb.AppendLine("        };");
        sb.AppendLine("    }");
    }

    private static void GenerateFromTargetMethod(StringBuilder sb, MapperInfo info)
    {
        sb.AppendLine($"    public static {info.SourceTypeName} From{info.TargetTypeName}(this {info.TargetTypeName} target)");
        sb.AppendLine("    {");
        sb.AppendLine("        if (target == null) throw new ArgumentNullException(nameof(target));");
        sb.AppendLine();
        sb.AppendLine($"        return new {info.SourceTypeName}");
        sb.AppendLine("        {");

        for (int i = 0; i < info.TargetMappings.Count; i++)
        {
            var mapping = info.TargetMappings[i];
            var assignment = GeneratePropertyAssignment(mapping);
            sb.Append($"            {mapping.TargetProperty} = {assignment}");
            
            if (i < info.TargetMappings.Count - 1)
                sb.AppendLine(",");
            else
                sb.AppendLine();
        }

        sb.AppendLine("        };");
        sb.AppendLine("    }");
    }

    private static string GeneratePropertyAssignment(PropertyMapping mapping)
    {
        return mapping.MappingType switch
        {
            MappingType.Direct => $"source.{mapping.SourceProperty}",
            MappingType.Convertible => $"({mapping.TargetType})source.{mapping.SourceProperty}",
            MappingType.Collection => $"source.{mapping.SourceProperty}?.ToList()",
            MappingType.Custom => $"MapCustom(source.{mapping.SourceProperty})",
            _ => $"source.{mapping.SourceProperty}"
        };
    }
}

// Data models
public record MapperInfo(
    string SourceTypeName,
    string SourceNamespace,
    string AccessModifier,
    string TargetTypeName,
    string TargetNamespace,
    bool GenerateReverseMapping,
    EquatableArray<PropertyMapping> SourceMappings,
    EquatableArray<PropertyMapping> TargetMappings);

public record PropertyMapping(
    string SourceProperty,
    string TargetProperty,
    string SourceType,
    string TargetType,
    MappingType MappingType);

public enum MappingType
{
    Direct,
    Convertible,
    Collection,
    Custom
}
```

## Example 3: Enum Extension Generator

Generate extension methods for enums with additional metadata.

### Attribute Definition

```csharp
[AttributeUsage(AttributeTargets.Field)]
public class EnumDisplayAttribute : Attribute
{
    public string DisplayName { get; }
    public string? Description { get; set; }
    public string? Category { get; set; }

    public EnumDisplayAttribute(string displayName)
    {
        DisplayName = displayName;
    }
}

[AttributeUsage(AttributeTargets.Enum)]
public class GenerateEnumExtensionsAttribute : Attribute
{
    public bool GenerateDisplayNames { get; set; } = true;
    public bool GenerateDescriptions { get; set; } = true;
    public bool GenerateCategories { get; set; } = false;
}
```

### Usage

```csharp
[GenerateEnumExtensions]
public enum Status
{
    [EnumDisplay("Not Started", Description = "Task has not been started")]
    NotStarted,
    
    [EnumDisplay("In Progress", Description = "Task is currently being worked on")]
    InProgress,
    
    [EnumDisplay("Completed", Description = "Task has been completed successfully")]
    Completed,
    
    [EnumDisplay("Cancelled", Description = "Task was cancelled")]
    Cancelled
}
```

### Generated Output

```csharp
public static class StatusExtensions
{
    public static string GetDisplayName(this Status status)
    {
        return status switch
        {
            Status.NotStarted => "Not Started",
            Status.InProgress => "In Progress",
            Status.Completed => "Completed",
            Status.Cancelled => "Cancelled",
            _ => status.ToString()
        };
    }

    public static string GetDescription(this Status status)
    {
        return status switch
        {
            Status.NotStarted => "Task has not been started",
            Status.InProgress => "Task is currently being worked on",
            Status.Completed => "Task has been completed successfully",
            Status.Cancelled => "Task was cancelled",
            _ => ""
        };
    }

    public static Status[] GetAllValues() => new[]
    {
        Status.NotStarted,
        Status.InProgress,
        Status.Completed,
        Status.Cancelled
    };

    public static bool TryParseDisplayName(string displayName, out Status status)
    {
        status = displayName switch
        {
            "Not Started" => Status.NotStarted,
            "In Progress" => Status.InProgress,
            "Completed" => Status.Completed,
            "Cancelled" => Status.Cancelled,
            _ => default
        };
        
        return displayName switch
        {
            "Not Started" or "In Progress" or "Completed" or "Cancelled" => true,
            _ => false
        };
    }
}
```

## Common Patterns

### 1. **Error Handling Pattern**

```csharp
private static void Execute(MyData? data, SourceProductionContext context)
{
    if (data is null) 
    {
        GeneratorService.Add(MyDiagnostics.InvalidData, Location.None);
        return;
    }

    try
    {
        var source = GenerateCode(data);
        context.AddSource($"{data.Name}.g.cs", source);
    }
    catch (Exception ex)
    {
        GeneratorService.Add(MyDiagnostics.GenerationError, Location.None);
    }
}
```

### 2. **Incremental Collection Pattern**

```csharp
private static EquatableArray<ItemInfo> GetItems(INamedTypeSymbol type)
{
    using var builder = ImmutableArrayBuilder<ItemInfo>.Rent();
    
    foreach (var member in type.GetMembers())
    {
        if (ShouldInclude(member))
        {
            builder.Add(CreateItemInfo(member));
        }
    }
    
    return builder.ToImmutable();
}
```

### 3. **Symbol Analysis Pattern**

```csharp
private static bool IsValidTarget(INamedTypeSymbol type)
{
    return type.IsPartial() &&
           !type.IsStatic &&
           type.DeclaredAccessibility == Accessibility.Public &&
           !type.IsAbstract;
}
```

### 4. **Attribute Processing Pattern**

```csharp
private static ConfigInfo ProcessAttribute(AttributeData attribute)
{
    var enabled = attribute.GetNamedArgument("Enabled", true);
    var mode = attribute.GetNamedArgument<GenerationMode>("Mode", GenerationMode.Default);
    var options = attribute.GetConstructorArguments<string>().ToArray();
    
    return new ConfigInfo(enabled, mode, options);
}
```

These examples demonstrate the power and flexibility of Six.GeneratorKit for creating sophisticated source generators with minimal boilerplate code.
